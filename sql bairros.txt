-- Criar nova tabela específica para as regiões dos bairros
CREATE TABLE IF NOT EXISTS regioes_bairro (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(50) NOT NULL UNIQUE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Criar tabela de Bairros apontando para a nova tabela
CREATE TABLE IF NOT EXISTS bairros (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    municipio VARCHAR(50) NOT NULL DEFAULT 'RECIFE',
    regiao_bairro_id INT NOT NULL,
    FOREIGN KEY (regiao_bairro_id) REFERENCES regioes_bairro(id) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
